# edit in place, save with Enter, focus returns to Edit
click:[data-fk="edit-9"]
expect-focus:#e-title
fill:#e-author=Unknown poet
press:#e-author=Enter

expect-focus:[data-fk="edit-9"]

# an edit that empties the title is refused and keeps the editor open
click:[data-fk="edit-9"]
fill:#e-title=

expect:#e-e-title=Give the book a title
click:[data-fk="e-cancel"]

# remove, then undo
click:[data-fk="remove-3"]

expect-focus:#pk-undo
expect-count:li.book=8

expect-count:li.book=9
